grammar Kubus;

program
    : namespace imports? classDef+ EOF
    ;

namespace
    : 'ns' qualifiedName ';'
    ;

imports
    : 'use' importList ';'
    ;

importList
    : qualifiedName (',' qualifiedName)*
    ;

classDef
    : qualifiedName (':' qualifiedName)? ('!' qualifiedName)? '{' member* '}'
    ;

member
    : visibility 'const' ID '=' expr ';'              # ConstantDecl
    | visibility ('static')? type '$' ID ('=' expr)? ';' # PropertyDecl
    | visibility ('static' | 'magic')? ID '(' paramList? ')' (type)? block # MethodDecl
    | '(' paramList? ')' block                       # ConstructorDecl
    | 'use' qualifiedName (',' qualifiedName)* ';'   # TraitUsage
    ;

visibility
    : 'public'
    | 'private'
    | 'protected'
    ;

paramList
    : param (',' param)*
    ;

param
    : '$' ID type
    ;

type
    : 'int'
    | 'float'
    | 'string'
    | 'bool'
    | 'void'
    | 'char'
    | ID
    ;

block
    : '{' statement* '}'
    ;

statement
    : variableDecl
    | localVariableDecl
    | staticLocalVariableDecl
    | returnStmt
    | ifStmt
    | foreachStmt
    | expr ';'                                        # ExpressionStmt
    ;

variableDecl
    : '$' ID ('[' expr? ']')? '=' expr ';'            # ArrayAppendStmt
    | '$' ID ('++' | '--') ';'                        # VariableIncDecStmt
    | '$' ID '=' expr ';'                             # VariableAssignStmt
    ;

localVariableDecl
    : '$$' ID ('++' | '--') ';'                       # LocalVariableIncDecStmt
    | '$$' ID '=' expr ';'                            # LocalVariableAssignStmt
    ;

staticLocalVariableDecl
    : '$$$' ID ('++' | '--') ';'                      # StaticLocalVariableIncDecStmt
    | '$$$' ID '=' expr ';'                           # StaticLocalVariableAssignStmt
    ;

returnStmt
    : 're' expr ';'
    ;

ifStmt
    : 'if' '(' expr ')' block ('else' block)?
    ;

foreachStmt
    : 'foreach' '(' expr ':' expr ')' block
    ;

expr
    : literal                                         # LiteralExpr
    | variable                                        # VariableExpr
    | expr ('++' | '--')                              # PostfixExpr
    | 'new' qualifiedName '(' (expr (',' expr)*)? ')' # NewExpr
    | '$$$' ID '(' (expr (',' expr)*)? ')'            # StaticFunctionCallExpr
    | ID '(' (expr (',' expr)*)? ')'                  # FunctionCallExpr
    | '(' expr ')'                                    # ParenExpr
    | '[' (expr (',' expr)*)? ']'                     # ArrayExpr
    | chain                                           # ChainExpr
    | '!' expr                                        # UnaryExpr
    | '-' expr                                        # UnaryExpr
    | expr ('*' | '/' | '%') expr                     # BinaryExpr
    | expr ('+' | '-') expr                           # BinaryExpr
    | expr ('>' | '>=' | '<' | '<=') expr             # BinaryExpr
    | expr ('==' | '!=') expr                         # BinaryExpr
    | expr '..' expr                                  # ConcatExpr
    ;

chain
    : (ID | variable) ('.' (ID | ID '(' (expr (',' expr)*)? ')'))* # QualifiedChain
    ;

literal
    : INTEGER
    | FLOAT
    | STRING
    | CHAR
    | BOOLEAN
    ;

variable
    : ('$$$' | '$$' | '$') ID
    ;

qualifiedName
    : ID ('\\' ID)*
    ;

ID: [a-zA-Z_][a-zA-Z0-9_]* ;
INTEGER: [0-9]+ ;
FLOAT: [0-9]+ '.' [0-9]+ ;
STRING: '"' (~["\\] | '\\' .)* '"' | '\'' (~['\\] | '\\' .)* '\'' ;
CHAR: '\'' . '\'' ;
BOOLEAN: 'true' | 'false' ;
WS: [ \t\r\n]+ -> skip ;
